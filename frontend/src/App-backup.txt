
import { useState } from 'react'
import './App.css'

function App() {
  const products = [
    {
      id: 1,
      name: 'Classic Formal Shirt',
      price: 2499,
      image: '/images/shirt.jpg',
      category: 'SMART CASUAL',
    },
    {
      id: 2,
      name: 'Essential Jacket',
      price: 4999,
      image: '/images/jacket.jpg',
      category: 'OUTERWEAR',
    },
    {
      id: 3,
      name: 'Classic Leather Shoes',
      price: 3499,
      image: '/images/shoes.jpg',
      category: 'FOOTWEAR',
    },
  ]

  const [cart, setCart] = useState([])
  const [search, setSearch] = useState('')
  const [discountCode, setDiscountCode] = useState('')
  const [appliedDiscount, setAppliedDiscount] = useState(false)
  const [discountMessage, setDiscountMessage] = useState('')

  function addToCart(product) {
    setCart((currentCart) => {
      const existingProduct = currentCart.find(
        (item) => item.id === product.id
      )

      if (existingProduct) {
        return currentCart.map((item) =>
          item.id === product.id
            ? { ...item, quantity: item.quantity + 1 }
            : item
        )
      }

      return [...currentCart, { ...product, quantity: 1 }]
    })
  }

  function changeQuantity(productId, amount) {
    setCart((currentCart) =>
      currentCart.map((item) =>
        item.id === productId
          ? {
              ...item,
              quantity: Math.max(1, item.quantity + amount),
            }
          : item
      )
    )
  }

  function removeFromCart(productId) {
    setCart((currentCart) =>
      currentCart.filter((item) => item.id !== productId)
    )

    setAppliedDiscount(false)
    setDiscountMessage('')
  }

  function applyDiscount() {
    if (cart.length === 0) {
      setAppliedDiscount(false)
      setDiscountMessage('Please add a product to your bag first.')
      return
    }

    if (discountCode.trim().toUpperCase() === 'URBAN10') {
      setAppliedDiscount(true)
      setDiscountMessage('10% discount applied successfully!')
    } else {
      setAppliedDiscount(false)
      setDiscountMessage('Invalid code. Please try URBAN10.')
    }
  }

  const filteredProducts = products.filter((product) =>
    `${product.name} ${product.category}`
      .toLowerCase()
      .includes(search.trim().toLowerCase())
  )

  const cartCount = cart.reduce(
    (sum, item) => sum + item.quantity,
    0
  )

  const subtotal = cart.reduce(
    (sum, item) => sum + item.price * item.quantity,
    0
  )

  const discountAmount = appliedDiscount ? subtotal * 0.1 : 0
  const finalTotal = subtotal - discountAmount

  function handleCheckout() {
    alert('Checkout will be available after backend integration.')
  }

  return (
    <div className="store">
      <div className="announcement">
        FREE SHIPPING ON ORDERS OVER Rs. 5,000
      </div>

      <header className="navbar">
        <h1 className="brand">
          URBAN CROWN<span>.</span>
        </h1>

        <nav>
          <a href="#home">Home</a>
          <a href="#collection">Collection</a>
          <a href="#about">About</a>
          <a href="#contact">Contact</a>
        </nav>

        <a className="cart" href="#cart">
          🛍 My Bag ({cartCount})
        </a>
      </header>

      <main>
        <section className="hero-section" id="home">
          <div className="hero-content">
            <p className="eyebrow">THE NEW MEN'S COLLECTION</p>

            <h2>
              Style That
              <br />
              Speaks For You.
            </h2>

            <p className="description">
              Discover modern menswear made for confidence,
              comfort and everyday elegance.
            </p>

            <a className="shop-button" href="#collection">
              EXPLORE COLLECTION →
            </a>
          </div>

          <div className="hero-art">
            <div className="art-circle"></div>
            <div className="fashion-icon">♟</div>
            <p>TIMELESS STYLE. MODERN MAN.</p>
          </div>
        </section>

        <section className="collection" id="collection">
          <p className="eyebrow">CURATED FOR YOU</p>
          <h2>Shop The Collection</h2>

          <p className="section-description">
            Essential pieces for every modern wardrobe.
          </p>

          <div className="search-container">
            <input
              className="search-input"
              type="search"
              placeholder="Search shirts, jackets, shoes..."
              value={search}
              onChange={(event) => setSearch(event.target.value)}
              aria-label="Search products"
            />
          </div>

          <div className="product-grid">
            {filteredProducts.length > 0 ? (
              filteredProducts.map((product) => (
                <article className="product-card" key={product.id}>
                  <div className="product-image">
                    <img src={product.image} alt={product.name} />
                  </div>

                  <p className="category">{product.category}</p>
                  <h3>{product.name}</h3>

                  <p className="price">
                    Rs. {product.price.toLocaleString('en-PK')}
                  </p>

                  <button onClick={() => addToCart(product)}>
                    ADD TO BAG
                  </button>
                </article>
              ))
            ) : (
              <p className="no-results">
                No products found. Try searching shirt, jacket or shoes.
              </p>
            )}
          </div>
        </section>

        <section className="cart-section" id="cart">
          <p className="eyebrow">YOUR SELECTION</p>
          <h2>Shopping Bag</h2>

          {cart.length === 0 ? (
            <p className="empty-cart">
              Your bag is empty. Add a product to get started!
            </p>
          ) : (
            <>
              <div className="cart-items">
                {cart.map((product) => (
                  <div className="cart-item" key={product.id}>
                    <span>{product.name}</span>

                    <span>
                      Rs. {product.price.toLocaleString('en-PK')} each
                    </span>

                    <div className="quantity-controls">
                      <button
                        type="button"
                        aria-label={`Decrease ${product.name} quantity`}
                        onClick={() => changeQuantity(product.id, -1)}
                      >
                        −
                      </button>

                      <span>{product.quantity}</span>

                      <button
                        type="button"
                        aria-label={`Increase ${product.name} quantity`}
                        onClick={() => changeQuantity(product.id, 1)}
                      >
                        +
                      </button>
                    </div>

                    <span>
                      Rs. {(product.price * product.quantity).toLocaleString('en-PK')}
                    </span>

                    <button
                      className="remove-button"
                      onClick={() => removeFromCart(product.id)}
                    >
                      Remove
                    </button>
                  </div>
                ))}
              </div>

              <div className="discount-section">
                <h3>Discount Code</h3>

                <div className="discount-form">
                  <input
                    type="text"
                    placeholder="Enter discount code"
                    value={discountCode}
                    onChange={(event) => {
                      setDiscountCode(event.target.value)
                      setAppliedDiscount(false)
                      setDiscountMessage('')
                    }}
                  />

                  <button type="button" onClick={applyDiscount}>
                    APPLY
                  </button>
                </div>

                {discountMessage && (
                  <p className="discount-message">
                    {discountMessage}
                  </p>
                )}
              </div>

              <h3 className="cart-total">
                Subtotal: Rs. {subtotal.toLocaleString('en-PK')}
              </h3>

              {appliedDiscount && (
                <p className="discount-total">
                  Discount: - Rs. {discountAmount.toLocaleString('en-PK')}
                </p>
              )}

              <h3 className="cart-total">
                Total: Rs. {finalTotal.toLocaleString('en-PK')}
              </h3>

              <button
                className="checkout-button"
                onClick={handleCheckout}
              >
                PROCEED TO CHECKOUT
              </button>
            </>
          )}
        </section>

        <section className="about" id="about">
          <p className="eyebrow">OUR PHILOSOPHY</p>
          <h2>Less, But Better.</h2>

          <p>
            At URBAN CROWN, we believe great style is not about
            following every trend. It is about finding your own look
            through timeless designs and everyday comfort.
          </p>
        </section>
      </main>

      <footer id="contact">
        <h2>
          URBAN CROWN<span>.</span>
        </h2>
        <p>Modern essentials. Timeless confidence.</p>
        <p>© 2026 URBAN CROWN. All rights reserved.</p>
      </footer>
    </div>
  )
}

export default App
